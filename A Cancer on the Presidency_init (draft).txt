// Apply a theme by name, then push the (possibly new) corrr into any header
// already on screen — corrr alone only affects the engine's FUTURE renders.
// Also syncs the picker dropdown(s), so a swap triggered by GAME EVENTS
// (Code 2's themeSwap, ported from 1996: End of History) shows up there too.
function _applyTheme(name) {
    _acopTheme = (name in ACOP_THEMES) ? name : 'A Cancer on the Presidency';
    try { ACOP_THEMES[_acopTheme](); } catch (e) {}
    document.querySelectorAll('.game_header').forEach(h => { h.innerHTML = corrr; });
    document.querySelectorAll('.acop-theme-select').forEach(s => { s.value = _acopTheme; });
    _syncLowFxBg();   // theme fns rewrite the body background — low mode re-overrides
}
// Code 2 runs in its own Function scope — expose the registry + applier so
// question events can swap themes (and authors can register themes) from there.
window.ACOP_THEMES = ACOP_THEMES;
window._acopApplyTheme = _applyTheme;

// Build the <select>: options from the registry, current selection marked,
// onchange → _applyTheme. Carries the stable `.acop-theme-select` class (the
// hook used for idempotence + tests) in BOTH mount modes below.
function _buildThemeSelect() {
    const sel = document.createElement('select');
    sel.className = 'acop-theme-select';
    Object.keys(ACOP_THEMES).forEach(name => {
        const o = document.createElement('option');
        o.value = name; o.textContent = name;
        if (name === _acopTheme) o.selected = true;
        sel.appendChild(o);
    });
    sel.addEventListener('change', () => _applyTheme(sel.value));
    return sel;
}

// Mount the picker. Preferred path — mirroring 1972: More Than Ever — is to
// REPLACE the host site's native theme <select id="themePicker"> in place, so
// it inherits the host's chrome styling and sits where players expect it. If
// the host doesn't supply that element (older/other hosts, the standalone
// showcase), fall back to a compact control pinned to the page's top-right.
// Mounted once (guarded on the shared `.acop-theme-select` class); relocating
// the picker is isolated to this one function.
function _mountThemePicker() {
    if (document.querySelector('.acop-theme-select')) return;   // already mounted (either mode)
    if (!document.body) return;

    // Preferred: hijack the host's native picker (MTE-style replaceWith).
    const host = document.getElementById('themePicker');
    if (host) {
        const sel = _buildThemeSelect();
        sel.id = 'themePicker';                 // keep the id so host CSS still styles it
        host.replaceWith(sel);
        _applyTheme(_acopTheme);                // apply current (default) once, like MTE
        return;
    }

    // Fallback: floating control, present on every screen, styled to match the mod.
    const wrap = document.createElement('div');
    wrap.id = 'acop-theme-picker';
    wrap.style.cssText = 'position:fixed;top:8px;right:10px;z-index:100000;'
        + "font-family:'TypewriterDry',monospace;font-size:11px;letter-spacing:.06em;"
        + 'background:rgba(32,27,18,0.82);color:#f0e6cf;border:1px solid #6b5a33;'
        + 'border-radius:3px;padding:3px 6px;display:flex;align-items:center;gap:6px;'
        + 'box-shadow:1px 2px 6px rgba(0,0,0,0.4);opacity:0.85;';
    const label = document.createElement('span');
    label.textContent = 'THEME';
    const sel = _buildThemeSelect();
    sel.id = 'acop-theme-select';
    sel.style.cssText = "font-family:'TypewriterDry',monospace;font-size:11px;"

    wrap.appendChild(label); wrap.appendChild(sel);
    document.body.appendChild(wrap);
}

    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.name = 'game_answers';
    checkbox.value = radio.value;

    svg.setAttribute('viewBox', '0 0 21 21');

    }

// ── Shared #game_window observer ──────────────────────────────────────────────
// A single MutationObserver fans out to every subscriber, replacing the several
// independent { childList, subtree } observers that previously each fired (and
// each ran their own DOM sweeps) on every mutation. Defined idempotently on
// window so Code 1 and Code 2 share one observer regardless of load order.
window.__onGameWindowMutation = window.__onGameWindowMutation || (function () {
    const subs = [];
    let _pending = false;
    const run = () => {
        _pending = false;
        for (let i = 0; i < subs.length; i++) { try { subs[i](); } catch (e) {} }
    };
    const obs = new MutationObserver(() => {
        if (_pending) return;
        _pending = true;
        requestAnimationFrame(run);
    });
    (function attach() {
        const gw = document.getElementById('game_window');
        if (gw) obs.observe(gw, { childList: true, subtree: true });
        else requestAnimationFrame(attach);
    })();
    return function (cb) { subs.push(cb); try { cb(); } catch (e) {} };
})();

// Shared drag dispatch: one pair of document listeners for ALL draggable panels

let __dragState = null;
document.addEventListener('mousemove', (e) => {
    const d = __dragState;
    if (!d) return;
    const nl = d.startLeft + e.clientX - d.startX;
    const nt = d.startTop  + e.clientY - d.startY;
    d.el.style.left = nl + 'px';
    d.el.style.top  = nt + 'px';
    // Pin follows from the same math — no offsetLeft/offsetWidth reads (forced
    // layout) on the move hot path; halfW was cached at mousedown. The style
    // observer still covers non-drag repositioning.
    d.pin.style.left = (nl + d.halfW - 10) + 'px';
    d.pin.style.top  = (nt - 20) + 'px';
});
document.addEventListener('mouseup', () => {
    const d = __dragState;
    if (!d) return;
    d.el.style.cursor = 'grab';
    d.savedPos.left = d.el.style.left;
    d.savedPos.top  = d.el.style.top;

});

function makeDraggable(elId, savedPos) {
    const el = document.getElementById(elId);
    if (!el || el.dataset.dragHooked) return;
    el.dataset.dragHooked = 'true';

    el.style.left = savedPos.left;
    el.style.top = savedPos.top;
    el.style.cursor = 'grab';

    // create pushpin (id'd so a recreated panel replaces its old pin instead of
    // orphaning it in the persistent container)
    document.getElementById(elId + '-pin')?.remove();
    const pin = document.createElement('div');
    pin.id = elId + '-pin';
    pin.className = 'map-pin';
    pin.innerHTML = `<svg xmlns='http://www.w3.org/2000/svg' width='20' height='28' viewBox='0 0 20 28'><circle cx='10' cy='8' r='7' fill='#cc2200' stroke='#881100' stroke-width='1.5'/><ellipse cx='10' cy='7' rx='3' ry='2.5' fill='#ff6644' opacity='0.5'/><rect x='9' y='14' width='2' height='12' rx='1' fill='#888'/></svg>`;
    const parent = document.getElementById('main_content_area') || el.parentElement;
    parent.appendChild(el);
    parent.appendChild(pin);
    pin.style.left = (parseInt(el.style.left) + el.offsetWidth / 2 - 10) + 'px';
    pin.style.top = (parseInt(el.style.top) - 20) + 'px';

    function updatePin() {
        pin.style.left = (el.offsetLeft + el.offsetWidth / 2 - 10) + 'px';
        pin.style.top = (el.offsetTop - 20) + 'px';
    }

    el.addEventListener('mousedown', (e) => {
        if (e.target.tagName === 'BUTTON') return;
        __dragState = {
            el, pin, savedPos, updatePin,
            startX: e.clientX, startY: e.clientY,
            startLeft: el.offsetLeft, startTop: el.offsetTop,
            halfW: el.offsetWidth / 2,
        };
        el.style.cursor = 'grabbing';
        e.preventDefault();
    });

    new MutationObserver(updatePin).observe(el, { attributes: true, attributeFilter: ['style'] });
}

// Aftermath: the final election map inherits every ground-burst scorch from
// the night (plus a smoke halo), so the damage assessment shows a pocked map.
// Downwind fallout plumes on the final damage map — a teardrop of radioactive
// fallout streaming EAST from each ground burst (west-to-east prevailing winds),
// rounded at the source and tapering to a point downwind, in semi-transparent
// powder red so overlapping plumes build into the denser regions (as on a real
// fallout dose map). Two bands per burst: a wide low-dose halo + a narrower
// high-dose core.
var _NUKE_FALLOUT_ON   = true;   // draw fallout plumes on the final damage-assessment map
var _NUKE_FALLOUT_LEN  = 16;     // plume length = scorch radius × this (bigger bursts reach further)
var _NUKE_FALLOUT_HALO = 'rgba(232, 104, 130, 0.13)'; // outer low-dose plume (semi-transparent powder red)
var _NUKE_FALLOUT_CORE = 'rgba(226, 52, 100, 0.30)';  // inner high-dose core
// Grey wash over the state colours on the final damage map, so the (red) fallout
// plumes read clearly against the census tints (esp. the red Injured states).
// A semi-transparent grey rect laid over the map UNDER the plumes — mutes the
// states + background toward grey; the plumes, drawn on top, keep their colour.
var _NUKE_STATE_WASH_ON = true;
var _NUKE_STATE_WASH    = 'rgba(122, 124, 128, 0.62)';  // higher alpha = greyer states (the map's own saturate(1.4) fights it, so it runs fairly strong)
// A teardrop path: rounded head at the burst (x,y), tapering to a point `len`
// to the EAST; `drift` nudges the tail up/down for a fanned, wind-blown look.
function __nukePlumePath(x, y, len, halfW, drift) {
    const tx = x + len, ty = y + (drift || 0);
    return 'M ' + x + ' ' + (y - halfW) +
        ' C ' + (x + len * 0.45) + ' ' + (y - halfW) +
        ', ' + (x + len * 0.85) + ' ' + (ty - halfW * 0.16) +
        ', ' + tx + ' ' + ty +
        ' C ' + (x + len * 0.85) + ' ' + (ty + halfW * 0.16) +
        ', ' + (x + len * 0.45) + ' ' + (y + halfW) +
        ', ' + x + ' ' + (y + halfW) +
        ' A ' + halfW + ' ' + halfW + ' 0 0 0 ' + x + ' ' + (y - halfW) + ' Z';
}
function __nukeAftermath(finalOn, svg) {
    if (!finalOn || !svg) return;
    const scorches = window.__nukeScorches || [];
    if (!scorches.length) return;
    const per = _NUKE_FALLOUT_ON ? 4 : 2;   // children drawn per burst
    const want = scorches.length * per + (_NUKE_STATE_WASH_ON ? 1 : 0);   // + the wash rect
    const old = svg.querySelector('#nuke-aftermath-layer');
    if (old && old.childElementCount === want) return;   // current
    if (old) old.remove();   // stale (recorded set changed) — rebuild
    const NS = 'http://www.w3.org/2000/svg';
    const g = document.createElementNS(NS, 'g');
    g.id = 'nuke-aftermath-layer';
    g.style.pointerEvents = 'none';
    // Grey wash FIRST (bottom of the aftermath group, which itself sits over the
    // state paths) so the plumes drawn after it stay vivid on top.
    if (_NUKE_STATE_WASH_ON) {
        const wash = document.createElementNS(NS, 'rect');
        let vx = -1000, vy = -1000, vw = 4000, vh = 4000;   // generous cover if no viewBox
        const vb = svg.viewBox && svg.viewBox.baseVal;
        if (vb && vb.width) { vx = vb.x; vy = vb.y; vw = vb.width; vh = vb.height; }
        wash.setAttribute('x', vx); wash.setAttribute('y', vy);
        wash.setAttribute('width', vw); wash.setAttribute('height', vh);
        wash.setAttribute('fill', _NUKE_STATE_WASH);
        wash.setAttribute('pointer-events', 'none');
        g.appendChild(wash);
    }
    scorches.forEach(s => {
        // fallout plume first, so the crater sits on top of its source
        if (_NUKE_FALLOUT_ON) {
            // Deterministic per-burst wind variation (hash of the coordinates,
            // NOT Math.random): the svg — and this layer with it — is rebuilt
            // on every visit to the final map, and a re-rolled drift made the
            // damage map's plume pattern change between viewings of the same war.
            const h = Math.sin(s.x * 12.9898 + s.y * 78.233) * 43758.5453;
            const drift = ((h - Math.floor(h)) - 0.5) * s.r * 2.2;
            const haloLen = s.r * _NUKE_FALLOUT_LEN, haloW = s.r * 1.5;

// Broadcast clock: the night runs 8:00 PM → 2:00 AM EST across the 480 ticks.
function __nukeClock(t) {
    const mins = 20 * 60 + Math.round(t * 0.75);
    const h24 = Math.floor(mins / 60) % 24, mm = mins % 60;
    const h12 = ((h24 + 11) % 12) + 1;
    return h12 + ':' + String(mm).padStart(2, '0') + ' ' + (h24 >= 12 ? 'PM' : 'AM');
}
// Gratuitous re-strikes: _NUKE_RESTRIKES extra warheads spread over the
// biggest states (national share ≥ 2.5%), allocated proportional to size, and
// landing at random through the rest of the night after each state declares.
// Purely theatrical — no tally effect. 0 = off.
var _NUKE_RESTRIKES = 20;
var _NUKE_RESTRIKE_DELAY = [3000, 45000];   // ms after the state's first salvo
var _NUKE_RESTRIKE_LINES = ['RE-STRIKE REPORTED OVER {C}', 'SECOND SUN OVER {C}', 'FURTHER DETONATIONS NEAR {C}'];
function __nukeRestrikeQuota(abbr) {
    try {
        const sj = window.campaignTrail_temp.states_json || [];
        let tot = 0, pv = 0, big = 0;
        for (const s of sj) { tot += s.fields.popular_votes || 0; }
        if (!tot) return 0;
        for (const s of sj) {
            const v = s.fields.popular_votes || 0;
            if (v / tot >= 0.025) big += v;
            if (s.fields.abbr === abbr) pv = v;
        }
        if (pv / tot < 0.025 || !big) return 0;
        return Math.round(_NUKE_RESTRIKES * pv / big);
    } catch (e) { return 0; }
}
// MIRVs: salvo size scales with the state's share of the national count —
// the big states eat two or three warheads.
function __nukeSalvoSize(abbr) {
    try {
        const sj = window.campaignTrail_temp.states_json || [];
        let tot = 0, pv = 0;
        for (const s of sj) {
            tot += s.fields.popular_votes || 0;
            if (s.fields.abbr === abbr) pv = s.fields.popular_votes || 0;
        }
        const share = tot > 0 ? pv / tot : 0;
        return share >= 0.05 ? 3 : share >= 0.025 ? 2 : 1;
    } catch (e) { return 1; }
}

// A state is called → its salvo. First warhead always ground-bursts on the
// centroid area (the scorch every state keeps); follow-ups spread across the
// state and may air-burst (flash + wide shockwave, no cloud, no scorch).
function __nukeStrikeAt(svg, path, censusColor) {
    // Paint the deferred census colour on the state (see the strike observer).
    const applyColor = () => { if (censusColor != null) { try { path.setAttribute('fill', censusColor); } catch (e) {} } };
    if (document.body.classList.contains('low-fx-on')) { applyColor(); return; }
    let bb;
    try { bb = path.getBBox(); } catch (e) { applyColor(); return; }
    if (!bb || (!bb.width && !bb.height)) { applyColor(); return; }
    const reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const g = __nukeStrikeLayer(svg);
    const abbr = __nukeMaps(svg).byNode.get(path) || null;
    const off = (abbr && _NUKE_HIT_OFFSET[abbr]) || { x: 0, y: 0 };

    // the whole "state falls" beat coincides with the impact, not the launch.

        __nukeDegrade();
    };
    // No missile flight under reduced motion → fire the beat immediately.
    if (reduce) onLeadImpact();
    for (let i = 0; i < n; i++) {
        const tx = bb.x + bb.width * (i === 0 ? 0.5 : 0.25 + Math.random() * 0.5) + off.x;
        const ty = bb.y + bb.height * (i === 0 ? 0.5 : 0.25 + Math.random() * 0.5) + off.y;
        const ground = i === 0 || Math.random() > 0.45;

        setTimeout(() => { try { __nukeOneMissile(svg, g, tx, ty, bb, ground, onImpact); } catch (e) {} }, delay);

    // normal one doesn't stack right after).
    if (abbr === 'DC') {
        setTimeout(() => { __nukeRollJudder(); __nukeJudder(true); __nukeStaticFlare(true); }, _NUKE_FLIGHT_MS);
    }
    // the gratuitous barrage: this state's share of the re-strikes, raining in
    // at random for the rest of the night (guards: screen gone, war unwound,
    // low demand toggled mid-night)
    const extra = abbr ? __nukeRestrikeQuota(abbr) : 0;
    for (let k = 0; k < extra; k++) {

            if (document.body.classList.contains('low-fx-on')) return;

            try { __nukeOneMissile(svg, __nukeStrikeLayer(svg), rx, ry, bb2, Math.random() > 0.5); } catch (e) {}

function __nukeOneMissile(svg, g, tx, ty, bb, ground, onImpact) {
    const NS = 'http://www.w3.org/2000/svg';
    const mk = (tag, attrs) => {
        const el = document.createElementNS(NS, tag);
        for (const k in attrs) el.setAttribute(k, attrs[k]);
        return el;
    };
    // Every ground burst is RECORDED for the final-map aftermath layer (always
    // — the damage-assessment map survives even with the live scorch off). The
    // LIVE persistent scorch circle on the Election Night map is separate and
    // optional (_NUKE_SCORCH, ships off): the state's own colour change is the
    // lasting mark there. Air bursts never scorch or record.
    const scorchR = Math.max(4, Math.min(bb.width, bb.height) * 0.14);
    let scorch = null;
    if (ground) {
        window.__nukeScorches = window.__nukeScorches || [];
        if (window.__nukeScorches.length < 400) window.__nukeScorches.push({ x: tx, y: ty, r: scorchR });
        if (_NUKE_SCORCH) {
            scorch = mk('circle', { cx: tx, cy: ty, r: scorchR, fill: '#140d08', opacity: '0' });
            g.appendChild(scorch);
        }
    }
    const noMotion = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    if (noMotion || !g.animate) { if (scorch) scorch.setAttribute('opacity', '0.45'); if (onImpact) onImpact(); return; }
    const FLIGHT = _NUKE_FLIGHT_MS;
    if (scorch) scorch.animate([{ opacity: 0 }, { opacity: 0.45 }], { duration: 400, delay: FLIGHT, fill: 'forwards' });
    // incoming: tracer line + the missile svg riding it. Approach vectors:
    // most drop in from the TOP of the map (ICBMs over the Arctic from the
    // USSR); a share come in from the RIGHT edge (Atlantic submarines).
    let edgeX = 960;
    try {
        const vb = svg.viewBox && svg.viewBox.baseVal;
        if (vb && vb.width) edgeX = vb.x + vb.width;
        else if (svg.width && svg.width.baseVal && svg.width.baseVal.value) edgeX = svg.width.baseVal.value;
    } catch (e) {}
    let sx, sy;
    if (Math.random() < _NUKE_SLBM_SHARE) {
        sx = edgeX + 40;
        sy = Math.max(-20, ty - (Math.random() * 120 + 40));   // shallow dive from off the right edge
    } else {
        sx = tx + (Math.random() * 240 - 120);                 // near-vertical, slight drift
        sy = -30;
    }
    const len = Math.hypot(tx - sx, ty - sy);
    const tracer = mk('line', { x1: sx, y1: sy, x2: tx, y2: ty, stroke: '#ffe2b0', 'stroke-width': '1.4', 'stroke-linecap': 'round', opacity: '0.85' });
    tracer.setAttribute('stroke-dasharray', String(len));
    tracer.setAttribute('stroke-dashoffset', String(len));
    g.appendChild(tracer);
    tracer.animate([{ strokeDashoffset: len }, { strokeDashoffset: 0 }], { duration: FLIGHT, easing: 'cubic-bezier(.55,0,1,.55)', fill: 'forwards' });
    const deg = Math.atan2(ty - sy, tx - sx) * 180 / Math.PI + _NUKE_MISSILE_UP;
    const mg = mk('g', {});
    const rot = mk('g', { transform: 'rotate(' + deg.toFixed(1) + ')' });
    const _ms = _NUKE_MISSILE_SIZE;
    const mimg = mk('image', { x: -_ms / 2, y: -_ms, width: _ms, height: _ms });
    mimg.setAttribute('href', _NUKE_MISSILE_IMG);
    rot.appendChild(mimg);
    mg.appendChild(rot);
    mg.style.transform = 'translate(' + sx + 'px,' + sy + 'px)';
    g.appendChild(mg);
    mg.animate([
        { transform: 'translate(' + sx + 'px,' + sy + 'px)' },
        { transform: 'translate(' + tx + 'px,' + ty + 'px)' },
    ], { duration: FLIGHT, easing: 'cubic-bezier(.55,0,1,.55)', fill: 'forwards' });
    // impact — air bursts flash bigger and ring wider but leave no cloud
    setTimeout(() => {
        mg.remove();
        if (onImpact) onImpact();   // the state turns its census colour AS the warhead lands
        __nukeRumble();

        tracer.animate([{ opacity: 0.85 }, { opacity: 0 }], { duration: 240, fill: 'forwards' });
        const flash = mk('circle', { cx: tx, cy: ty, r: 3, fill: '#fff7dd', opacity: '0.95' });

        cloud.style.opacity = '0';
        g.appendChild(cloud);
        // _NUKE_CLOUD_SCALE sizes the whole cloud (procedural or custom image).
        const _cs = _NUKE_CLOUD_SCALE;
        cloud.animate([
            { opacity: 0,    transform: 'translateY(2px) scale(' + (0.4 * _cs) + ')' },

            { opacity: 0.85, transform: 'translateY(-2px) scale(' + (1 * _cs) + ')',   offset: 0.55 },
            { opacity: 0,    transform: 'translateY(-5px) scale(' + (1.08 * _cs) + ')' },
        ], { duration: _NUKE_IMPACT_MS, easing: 'ease-out', fill: 'forwards' });
        setTimeout(() => { flash.remove(); ring.remove(); cloud.remove(); tracer.remove(); }, _NUKE_IMPACT_MS + 200);
    }, FLIGHT);
}
function __nukeStrikeWatch(nukeOn) {
    const night = nukeOn && __isElectionNightRaw();
    const svg = night ? document.querySelector('#map_container svg') : null;
    if (svg === __nukeStrikeSvg) return;   // already watching exactly this svg (or already off)
    if (__nukeStrikeObs) { try { __nukeStrikeObs.disconnect(); } catch (e) {} __nukeStrikeObs = null; }
    __nukeStrikeSvg = svg;
    if (!svg) return;
    __nukeRollJudder();   // fresh night → fresh judder schedule
    __nukeEbsIntroPlayed = false; __nukeEbsIntroArmed = false;   // …and re-arm the EBS intro
    const censusFills = (window.__nukeCensusColors || []).map(__nukeCanonColor);
    __nukeStrikeObs = new MutationObserver(muts => {
        for (const m of muts) {
            if (m.type !== 'attributes' || m.attributeName !== 'fill') continue;
            const p = m.target;
            if (!p || String(p.tagName).toLowerCase() !== 'path') continue;
            const oldV = __nukeCanonColor(m.oldValue || '');
            const newV = __nukeCanonColor(p.getAttribute('fill') || '');
            if (newV === oldV) continue;                          // hover re-sets etc.
            if (censusFills.length && censusFills.indexOf(newV) === -1) continue;   // not a call
            if (censusFills.indexOf(oldV) !== -1) continue;       // already called (shade tweak)
            if (p.__nukeStruck) continue;                         // our own at-impact repaint
            p.__nukeStruck = true;
            // Defer the state's colour to the missile's impact: revert it to the
            // pre-call fill now and hand the census colour to __nukeStrikeAt to
            // paint when the warhead lands (so the map goes red as it's hit, not
            // ~1.4s before). Only when we can actually revert — otherwise fall
            // back to the old immediate recolour.
            let censusColor = null;
            if (m.oldValue != null) {
                censusColor = p.getAttribute('fill');
                try { p.setAttribute('fill', m.oldValue); } catch (e) { censusColor = null; }
            }
            __nukeStrikeAt(svg, p, censusColor);
        }
    });
    __nukeStrikeObs.observe(svg, { attributes: true, subtree: true, attributeFilter: ['fill'], attributeOldValue: true });
}

// Election Night + final election map: dress the electoral-vote readout
// (#overall_result) as the 1976 ACOP broadcast scoreboard — rebuild each <li>
// ("Name: EV / PV%") into name/num spans carrying the candidate's colour as
// --pill-color, and append an "ACOP News Projection" footer. Styling
// (gold-framed recessed pills, embossed glow, CRT scanlines) is the
// .acop-final-tv CSS in logo-hide-style above.
// Idempotent PER <li>, not per panel: Election Night rewrites the <ul> every
// ~2s as states are called (a one-shot flag would leave the list plain after
// the first engine update), so each tick re-dresses only unclassed rows.
function __scoreboardTick() {
    if (!__isFinalElectionMap() && !__isElectionNight()) return;
    const box = document.getElementById('overall_result');
    if (!box) return;
    const lis = box.querySelectorAll('ul > li:not(.acop-score-pill)');
    if (!lis.length) return;
    lis.forEach(li => {
        const swatch = li.querySelector('span');
        const color = swatch ? getComputedStyle(swatch).backgroundColor : '';
        if (swatch) swatch.remove();
        const raw = li.textContent.trim();               // "Ford: 160 / 47.9%"
        const ci = raw.indexOf(':');
        const name = (ci >= 0 ? raw.slice(0, ci) : raw).trim();
        const rest = (ci >= 0 ? raw.slice(ci + 1) : '').trim();  // "160 / 47.9%"
        const num = (rest.split('/')[0] || rest).trim();  // "160"  (PV%-only runs keep the %)
        li.classList.add('acop-score-pill');
        if (color) li.style.setProperty('--pill-color', color);
        li.textContent = '';                              // clear, then rebuild via textContent (no injection)
        const n = document.createElement('span'); n.className = 'acop-pill-name'; n.textContent = name;
        const v = document.createElement('span'); v.className = 'acop-pill-num';  v.textContent = num;
        li.appendChild(n); li.appendChild(v);
    });
    if (!box.querySelector('.acop-proj')) {
        // A <div>, NOT a <p>: Election Night's updater rewrites every direct
        // child p ($('#overall_result > p').html('X% complete…')) each tally
        // tick — a <p> footer would be overwritten with the percentage text.
        const proj = document.createElement('div');
        proj.className = 'acop-proj';
        proj.textContent = 'ACOP News Projection';
        box.appendChild(proj);
    }
}

// (#state_result) as the ACOP '76 state graphic — "EV / STATE / PRESIDENT"

var _ACOP_PARTY_ABBR = {
    'republican': 'REP.', 'democratic': 'DEM.', 'democrat': 'DEM.',
    'libertarian': 'LIB.', 'independent': 'IND.', 'the fourth estate': 'PRESS'
};
function __statePanelTick() {
    if (!__isFinalElectionMap() && !__isElectionNight()) return;
    const box = document.getElementById('state_result');
    if (!box || box.querySelector('.acop-sr-head') || !box.querySelector('ul')) return;
    const ct = window.campaignTrail_temp;
    if (!ct || !ct.states_json || !ct.final_state_results || !ct.candidate_json) return;
    const nameP = box.querySelector('p');
    const stateName = nameP ? nameP.textContent.trim() : '';
    const stateObj = ct.states_json.find(s => s.fields && s.fields.name === stateName);
    if (!stateObj) return;
    const sr = ct.final_state_results.find(r => String(r.state) === String(stateObj.pk));
    if (!sr || !sr.result) return;
    // mirror the engine's own row selection exactly (same slice, same filter,
    // same order) so the card always shows what the plain panel would have
    const rows = sr.result.slice(0, 4).filter(r => r.percent > 0);
    if (!rows.length) return;

    const el = (tag, cls, text) => {
        const n = document.createElement(tag);
        n.className = cls;
        if (text != null) n.textContent = text;   // textContent only — no injection
        return n;
    };
    const head = el('div', 'acop-sr-head');
    const evs = Number(stateObj.fields.electoral_votes) || 0;
    if (evs > 0) {
        const evBox = el('div', 'acop-sr-evs');
        evBox.appendChild(el('span', 'acop-sr-evnum', String(evs)));
        evBox.appendChild(el('span', 'acop-sr-evlbl', evs === 1 ? 'EV' : 'EVS'));
        head.appendChild(evBox);
    }
    const title = el('div', 'acop-sr-title');
    title.appendChild(el('span', 'acop-sr-state', stateName.toUpperCase()));
    title.appendChild(el('span', 'acop-sr-office', 'PRESIDENT'));
    head.appendChild(title);

    const panels = el('div', 'acop-sr-panels');
    const main = el('div', 'acop-sr-main');
    const side = el('div', 'acop-sr-side');
    rows.forEach(r => {
        const cand = ct.candidate_json.find(c => String(c.pk) === String(r.candidate));
        const f = (cand && cand.fields) || {};
        const row = el('div', 'acop-sr-row');
        row.appendChild(el('div', 'acop-sr-name', (f.last_name || '—').toUpperCase()));
        row.appendChild(el('div', 'acop-sr-votes', Math.round(r.votes || 0).toLocaleString('en-US')));
        main.appendChild(row);
        const prow = el('div', 'acop-sr-prow');
        prow.appendChild(el('div', 'acop-sr-party',
            _ACOP_PARTY_ABBR[String(f.party || '').toLowerCase()] || 'IND.'));
        // broadcast-style integers, but a candidate with real votes should
        // never read "0%" — anything that rounds to zero shows as <1%
        const p100 = 100 * (r.percent || 0);
        prow.appendChild(el('div', 'acop-sr-pct', p100 < 0.5 ? '<1%' : Math.round(p100) + '%'));
        side.appendChild(prow);
    });
    panels.appendChild(main);
    panels.appendChild(side);

    box.textContent = '';
    box.appendChild(head);
    box.appendChild(panels);
}

    // TUNING (in reel-body coordinates, like the tapes' left/top):
    //   pin         — where the tape meets the black pin
    //   angleOffset — extra degrees applied after the tangent solve (art fudge)
    //   radiusPad   — px added to the roll radius before solving (art fudge)
    // Live tools (browser console):
    //   __tapeStripTool()            click the LEFT pin, then the RIGHT pin —
    //                                geometry updates live, config printed for
    //                                pasting back into _TAPE_GEO below
    //   __tapeStripTune({left:{pin:{x,y},angleOffset,radiusPad}, right:{…}})
    //                                merge any subset, updates live
    var _TAPE_GEO = {
        left:  { pin: { x: 124, y: 265 }, angleOffset: 0, radiusPad: 0 },
        right: { pin: { x: 371, y: 274 }, angleOffset: 0, radiusPad: 0 },
    };

    // outerSign picks which of the two tangents from the pin we take: +1 walks
    // the tangent point to the LEFT side of the roll, -1 to the RIGHT.
    function _placeTapeStrip(el, cx, cy, r, geo, outerSign) {
        const px = geo.pin.x, py = geo.pin.y;
        const dx = px - cx, dy = py - cy;
        const d = Math.hypot(dx, dy);
        const rr = Math.max(1, r + (geo.radiusPad || 0));
        if (d <= rr + 0.5) { el.style.height = '0px'; return; }   // pin inside the roll — hide
        const base = Math.atan2(dy, dx);            // centre → pin direction
        const swing = Math.acos(rr / d);            // half-angle subtended by the tangents
        const g = base + outerSign * swing;
        const tx = cx + rr * Math.cos(g);           // tangent point on the roll edge
        const ty = cy + rr * Math.sin(g);
        const h = Math.hypot(px - tx, py - ty);
        // CSS rotate about 'top center': (0,h) → (-h·sinθ, h·cosθ); solve so the
        // strip's bottom end lands on the pin.
        const theta = Math.atan2(tx - px, py - ty) * 180 / Math.PI + (geo.angleOffset || 0);
        el.style.left = (tx - 1.5) + 'px';          // 3px-wide strip, pivot at its top centre
        el.style.top = ty + 'px';
        el.style.height = h + 'px';
        el.style.transform = 'rotate(' + theta + 'deg)';
    }

    // Tape strip + tape resize
    function updateTapeVisuals() {
    // Pre-play there is no duration yet — stay hidden, EXCEPT while the pin
    // tool is active (it needs the strips visible to aim by, so draw at 0%).
    if ((!audio.duration || isNaN(audio.duration)) && !window.__tapeToolActive) return;
    const pct = (!audio.duration || isNaN(audio.duration)) ? 0 : audio.currentTime / audio.duration * 100;

    const maxTape = 210, minTape = 10;
    const tapeRange = maxTape - minTape;
    const leftSize = maxTape - (tapeRange * pct / 100);
    const rightSize = minTape + (tapeRange * pct / 100);

    // Fixed centre points for each tape
    const leftCX = 125, leftCY = 127;
    const rightCX = 378, rightCY = 127;

    leftTape.style.width = leftSize + "px";
    leftTape.style.height = leftSize + "px";
    leftTape.style.left = (leftCX - leftSize / 2) + "px";
    leftTape.style.top = (leftCY - leftSize / 2) + "px";

    rightTape.style.width = rightSize + "px";
    rightTape.style.height = rightSize + "px";
    rightTape.style.left = (rightCX - rightSize / 2) + "px";
    rightTape.style.top = (rightCY - rightSize / 2) + "px";

    // Strips: tangent from each pin to its roll (see _placeTapeStrip).
    _placeTapeStrip(leftStrip,  leftCX,  leftCY,  leftSize / 2,  _TAPE_GEO.left,  +1);
    _placeTapeStrip(rightStrip, rightCX, rightCY, rightSize / 2, _TAPE_GEO.right, -1);
    }

    // Console tuning tools (see the _TAPE_GEO comment).
    window.__tapeStripTune = (cfg) => {
        ['left', 'right'].forEach(side => {
            const c = cfg && cfg[side];
            if (!c) return;
            if (c.pin) Object.assign(_TAPE_GEO[side].pin, c.pin);
            if ('angleOffset' in c) _TAPE_GEO[side].angleOffset = c.angleOffset;
            if ('radiusPad' in c) _TAPE_GEO[side].radiusPad = c.radiusPad;
        });
        updateTapeVisuals();
        return JSON.parse(JSON.stringify(_TAPE_GEO));
    };
    // Pin-placement tool. The recorder is normally a drag surface (cursor:grab,
    // mousedown moves it), so while the tool is active it takes the mouse over
    // completely: the drag handler stands down (__tapeToolActive guard there),
    // mousedown is swallowed in capture phase, and every click just places a
    // pin. Works before any music has played (strips draw at 0%). Esc cancels.
    window.__tapeStripTool = () => {
        const body = document.getElementById('reel-body');
        if (!body || !tapeRecorder) { console.warn('tape recorder not on screen'); return; }
        if (window.__tapeToolActive) { console.warn('[tape tool] already active'); return; }
        window.__tapeToolActive = true;
        const prevCursor = tapeRecorder.style.cursor;
        tapeRecorder.style.cursor = 'crosshair';
        const hint = document.createElement('div');
        hint.style.cssText = 'position:absolute;top:-26px;left:0;background:#111;color:#ffd24a;'
            + "font:11px 'TypewriterDry',monospace;letter-spacing:1px;padding:3px 9px;"
            + 'z-index:10001;white-space:nowrap;pointer-events:none;';
        hint.textContent = 'TAPE TOOL — click the LEFT pin (Esc cancels)';
        tapeRecorder.appendChild(hint);
        updateTapeVisuals();   // make the strips visible to aim by, even pre-play
        let stage = 0; // 0 = expect left-pin click, 1 = right-pin click
        const swallow = (e) => { e.preventDefault(); e.stopPropagation(); };
        const finish = () => {
            window.__tapeToolActive = false;
            tapeRecorder.style.cursor = prevCursor || 'grab';
            tapeRecorder.removeEventListener('mousedown', swallow, true);
            tapeRecorder.removeEventListener('click', onClick, true);
            document.removeEventListener('keydown', onKey, true);
            hint.remove();
        };
        const onClick = (e) => {
            e.preventDefault(); e.stopPropagation();
            const r = body.getBoundingClientRect();
            const x = Math.round(e.clientX - r.left), y = Math.round(e.clientY - r.top);
            if (stage === 0) {
                _TAPE_GEO.left.pin = { x, y };
                stage = 1;
                hint.textContent = 'left pin (' + x + ', ' + y + ') — now click the RIGHT pin';
                console.log('[tape tool] left pin = { x: ' + x + ', y: ' + y + ' }');
            } else {
                _TAPE_GEO.right.pin = { x, y };
                console.log('[tape tool] right pin = { x: ' + x + ', y: ' + y + ' }. Paste into _TAPE_GEO:');
                console.log(JSON.stringify(_TAPE_GEO));
                hint.textContent = 'done — config printed to console';
                setTimeout(finish, 1500);
            }
            updateTapeVisuals();
        };
        const onKey = (e) => { if (e.key === 'Escape') finish(); };
        // Capture phase: for clicks on any child of the recorder these run
        // BEFORE the recorder's own drag/button handlers and stop them.
        tapeRecorder.addEventListener('mousedown', swallow, true);
        tapeRecorder.addEventListener('click', onClick, true);
        document.addEventListener('keydown', onKey, true);
    };

    let visualTimer = null;
    function startVisuals(state) {
        if (visualTimer) clearInterval(visualTimer);
        setReelAnimation(state);
        visualTimer = setInterval(updateTapeVisuals, 100);
    }
    function stopVisuals() {
        if (visualTimer) clearInterval(visualTimer);
        visualTimer = null;
        setReelAnimation('pause');
    }

    // Playback state
    let playbackState = 'pause';
    let isPlaying = false;

    // ── Reactive soundtrack state (see SOUNDTRACK_PLAN.md) ──────────────────
    // Config (window.SOUNDTRACK: section lists, mood rules) and triggers live
    // in Code 2; these are the player-side primitives.
    let _targetVolume = 0.5;    // the user's chosen volume; fades always return here
    let _fadeTimer = null;
    let _pendingPlaylist = null; // section list queued behind the current song
    let _musicSection = 0;       // 0 = intro music, no section entered yet
    let _stingerCued = false;    // stinger loaded while paused; plays on resume
    const FADE_MS = 1200;

    function _fadeTo(vol, ms, cb) {
        if (_fadeTimer) { clearInterval(_fadeTimer); _fadeTimer = null; }
        const from = audio.volume, delta = vol - from;
        if (ms <= 0 || Math.abs(delta) < 0.01) {
            audio.volume = vol;
            if (cb) cb();
            return;
        }
        const stepMs = 50, steps = Math.max(1, Math.round(ms / stepMs));
        let i = 0;
        _fadeTimer = setInterval(() => {
            i++;
            audio.volume = Math.min(1, Math.max(0, from + delta * (i / steps)));
            if (i >= steps) {
                clearInterval(_fadeTimer);
                _fadeTimer = null;
                if (cb) cb();
            }
        }, stepMs);
    }

  function loadAndPlay() {
        if (window._introSound) {
            window._introSound.pause();
            window._introSound.currentTime = 0;
        }
        _stingerCued = false;
        audio.src = playlist.getCurrentSong().getAudioLink();
        updateUI();
        audio.play();
        playbackState = 'play';
        playBtn.classList.add('active');
        startVisuals('play');
        showStrips();
    }

   function showStrips() {
    // Heights (and angles) come from the tangent solve now — just recompute.
    // Before audio metadata arrives updateTapeVisuals no-ops and the strips
    // stay hidden (height 0); the 100ms visuals timer picks them up as soon
    // as duration is known.
    updateTapeVisuals();
}

    // Button events
playBtn.addEventListener('click', () => {
        // Stop any playing national anthem and restore main audio before toggling play/pause
        if (window.__activeAnthemStop) { window.__activeAnthemStop(false); audio.muted = false; }
        if (playbackState !== 'pause') {
            audio.pause();
            playbackState = 'pause';
            playBtn.classList.remove('active');
            stopVisuals();
        } else {
            // A section list queued while paused installs now (unless a stinger
            // was cued while paused — that plays first; the list installs on its
            // 'ended'). See SOUNDTRACK_PLAN.md §7 paused-at-boundary.
            if (_pendingPlaylist && !_stingerCued) { _installPending(); return; }
            if (window._introSound) {
                window._introSound.pause();
                window._introSound.currentTime = 0;
            }
            audio.play();
            playbackState = 'play';
            playBtn.classList.add('active');
            startVisuals('play');
            showStrips();
        }
    });

    // Shared seek transport: one interval, stopped on mouseup ANYWHERE. Previously
    // each press created its own interval cleared only by the button's own mouseup,
    // so releasing off-button left it seeking forever and rapid presses stacked.
    let _seekInterval = null;
    function _stopSeek() {
        if (!_seekInterval) return;
        clearInterval(_seekInterval);
        _seekInterval = null;
        if (isPlaying) {
            audio.play();
            playbackState = 'play';
            setReelAnimation('play');

            setReelAnimation('pause');
        }
    }

    function _rebuildPlaylistSelect() {
        // Single source of truth for the dropdown — route ALL playlist
        // mutations through this (the options are otherwise built once).
        playlistSelect.innerHTML = '';

    }

    function _installPending() {
        if (!_pendingPlaylist) return;
        playlist.songs = _pendingPlaylist.slice();
        playlist.currentSongIndex = 0;
        _pendingPlaylist = null;
        _stingerCued = false;
        _rebuildPlaylistSelect();
        loadAndPlay();
    }

    function _setCurrent(index) {
        // Load a track WITHOUT starting playback (loadAndPlay always plays):
        // used to cue a stinger while the player is paused/off.
        playlist.currentSongIndex = index;
        audio.src = playlist.getCurrentSong().getAudioLink();
        updateUI();
        playlistSelect.value = index;
    }

    window.__injectNextSong = function (song) {
        // Scripted question stinger: splice as the next track; fade-and-play
        // only if the tape is actually playing — never restart a player the
        // user switched off (locked decision).
        if (!song || !window.playlist) return;
        const at = playlist.currentSongIndex + 1;
        playlist.songs.splice(at, 0, song);
        _rebuildPlaylistSelect();
        // A stinger outranks a hovering national anthem: stop it and take the
        // tape back before fading, so the two never fight over the main audio.
        if (window.__activeAnthemStop) {
            window.__activeAnthemStop(false);
            audio.muted = false;
        }
        if (playbackState === 'play') {
            _fadeTo(0, FADE_MS, () => {
                playlist.currentSongIndex = at;
                loadAndPlay();
                _fadeTo(_targetVolume, FADE_MS);
            });
        } else {
            _setCurrent(at);
            _stingerCued = true;
        }
    };

    window.__enterMusicSection = function (n) {
        // Section boundary: QUEUE the section's list — the current song plays
        // out first, then the new list installs on 'ended'. Mood rules are read
        // here and only here (once per section, not per turn); passing rules'
        // tracks lead the section.
        if (!window.SOUNDTRACK || !window.playlist) return;
        if (!n || n === _musicSection) return; // idempotent — retry ticks re-hit the same pk
        _musicSection = n;
        const secDef = window.SOUNDTRACK.sections && window.SOUNDTRACK.sections[n];
        let base = [];
        try { base = (secDef && secDef.tracks) ? secDef.tracks() : []; } catch (err) {}
        let mood = [];
        (window.SOUNDTRACK.mood || []).forEach(rule => {
            try { if (rule.when()) mood = mood.concat(rule.tracks() || []); } catch (err) {}
        });
        const list = mood.concat(base).filter(Boolean);
        if (!list.length) return; // section has no tracks authored — keep the current list
        _pendingPlaylist = list;
        // If nothing is playing, 'ended' never fires — the list installs on the
        // next play/next press instead (see those handlers).
    };

    window.__resetSoundtrack = function () {
        // New Game: clear section tracking + queued list. Deliberately does NOT
        // rebuild the playlist or touch playback (respects a switched-off
        // player; avoids yanking music mid-track).
        _musicSection = 0;
        _pendingPlaylist = null;
        _stingerCued = false;
    };

    window._rebuildPlaylistSelect = _rebuildPlaylistSelect;

// ── EDUCATIONAL TOOLTIPS ──────────────────────────────────────────────────────
// A second class of contextual tooltips: historical/civics context, constant
// across the whole game (which is why they live here in Code 1, not with the
// people-tooltips in Code 2). Same card look, NO image area, and the trigger
// word gets a DOUBLE pen-stroke underline instead of a single one.
//
// Toggle: an "Educational tooltips" checkbox appears next to the difficulty
// selector before the game begins (state remembered in localStorage).
//
// TO ADD A TERM just append an entry here. searchStrings are matched
// case-insensitively on word edges; put longer variants first for clarity but
// order doesn't matter (longest-at-position wins automatically).
var EDU_TOOLTIPS = [
    {
        searchStrings: ['impoundment', 'impound'],
        label: 'Impoundment',
        body: 'The practice of not spending funds appropriated by Congress. It has been going on since Jefferson but critics say that it unconstitutionally breaks the power of the purse, particularly when used for ideological purposes.'
    },
];
// ── end of the editable list — machinery below ───────────────────────────────

var _eduTipsOn = false;   // OFF by default — opt in via the checkbox
try { _eduTipsOn = localStorage.getItem('acop_edu_tips') === '1'; } catch (e) {}
if (document.body) document.body.classList.toggle('edu-tips-off', !_eduTipsOn);

// Single THICK pen stroke — same hand-drawn language as Code 2's
// generatePenStroke (2.1px) but at roughly double the weight, so the two
// tooltip classes stay distinguishable at a glance.
var _EDU_STROKE =
    '<svg class="edu-stroke" style="position:absolute;bottom:-1px;left:0;width:100%;height:6px;overflow:visible;pointer-events:none" preserveAspectRatio="none" viewBox="0 0 100 6">' +
    // Same squiggle as the people-tips pen stroke but with the wobble damped to
    // half its deviation from the baseline — reads straighter, still hand-drawn.
    '<path d="M 0 0 Q 35 3.2 50 2.5 Q 65 1.8 100 4" fill="none" stroke="rgb(96, 115, 129)" stroke-width="3.3" vector-effect="non-scaling-stroke"/>' +
    '</svg>';

// Matchers: one case-insensitive word-edge regex per searchString.
var _eduMatchers = [];
EDU_TOOLTIPS.forEach(entry => {
    (entry.searchStrings || []).forEach(s => {
        const esc = s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
        _eduMatchers.push({ re: new RegExp('(?<!\\w)' + esc + '(?!\\w)', 'gi'), entry });
    });
});

function _eduSpan(matchedText, entry) {
    const s = document.createElement('span');
    s.className = 'edu-tip';
    s.style.position = 'relative';
    s.appendChild(document.createTextNode(matchedText));
    s.insertAdjacentHTML('beforeend',
        _EDU_STROKE +
        '<span class="edu-tiptext"><span class="tt-label"></span>' +
        '<span class="tt-body"><span class="tt-body-text"></span></span></span>');
    s.querySelector('.tt-label').textContent = entry.label;
    s.querySelector('.tt-body-text').textContent = entry.body;
    return s;
}

// Wrap matches within one text node; returns a fragment or null if no match.
function _eduWrapText(text) {
    let rest = text, found = false;
    const frag = document.createDocumentFragment();
    for (;;) {
        let best = null;
        for (const m of _eduMatchers) {
            m.re.lastIndex = 0;
            const x = m.re.exec(rest);
            if (x && (!best || x.index < best.index ||
                     (x.index === best.index && x[0].length > best.text.length))) {
                best = { index: x.index, text: x[0], entry: m.entry };
            }
        }
        if (!best) break;
        found = true;
        if (best.index > 0) frag.appendChild(document.createTextNode(rest.slice(0, best.index)));
        frag.appendChild(_eduSpan(best.text, best.entry));
        rest = rest.slice(best.index + best.text.length);
    }
    if (!found) return null;
    if (rest) frag.appendChild(document.createTextNode(rest));
    return frag;
}

// Bake .edu-tip spans into an HTML string. Called by Code 2's tooltip pipeline
// BEFORE its own people-matcher runs (so that matcher sees our tags and skips
// them). DOM-walking text nodes means tags/attributes can never be corrupted,
// and skipping .edu-tip/.mytooltip subtrees makes the pass idempotent.
window.applyEduTooltips = function (html) {
    if (typeof html !== 'string' || !html || !_eduMatchers.length) return html;
    let any = false;
    for (const m of _eduMatchers) { m.re.lastIndex = 0; if (m.re.test(html)) { any = true; break; } }
    if (!any) return html;   // common case: leave the string byte-identical
    const tpl = document.createElement('template');
    tpl.innerHTML = html;
    let changed = false;
    (function walk(node) {
        Array.from(node.childNodes).forEach(ch => {
            if (ch.nodeType === 1) {
                const cl = ch.classList;
                if (cl.contains('edu-tip') || cl.contains('mytooltip') || cl.contains('mytooltiptext') || cl.contains('xpl')) return;
                walk(ch);
            } else if (ch.nodeType === 3) {
                const frag = _eduWrapText(ch.nodeValue);
                if (frag) { ch.replaceWith(frag); changed = true; }
            }
        });
    })(tpl.content);
    return changed ? tpl.innerHTML : html;
};

    _EXPLETIVES.map(w => w.replace(/[.*+?^${}()|[\]\\]/g, '\\$&').replace(/ /g, '\\s+')).join('|') +
    ')(?![\\w-])', 'gi');

function _explWrapText(text) {
    _EXPL_RE.lastIndex = 0;
    if (!_EXPL_RE.test(text)) return null;
    _EXPL_RE.lastIndex = 0;
    const frag = document.createDocumentFragment();
    let last = 0, m;
    while ((m = _EXPL_RE.exec(text)) !== null) {
        if (m.index > last) frag.appendChild(document.createTextNode(text.slice(last, m.index)));
        const wrap = document.createElement('span');
        wrap.className = 'xpl';
        const raw = document.createElement('span');
        raw.className = 'xpl-w';
        raw.textContent = m[0];                       // original word, case preserved
        const clean = document.createElement('span');
        clean.className = 'xpl-c';
        clean.textContent = '[EXPLETIVE DELETED]';
        wrap.appendChild(raw); wrap.appendChild(clean);
        frag.appendChild(wrap);
        last = m.index + m[0].length;
    }
    if (last < text.length) frag.appendChild(document.createTextNode(text.slice(last)));
    return frag;
}

// Bake .xpl spans. Runs in Code 2's pipeline; DOM-walks text nodes (tags/
// attributes never corrupted) and skips existing tip/xpl subtrees (idempotent).
window.applyExpletiveMarkup = function (html) {
    if (typeof html !== 'string' || !html) return html;
    _EXPL_RE.lastIndex = 0;
    if (!_EXPL_RE.test(html)) return html;            // common case: byte-identical
    const tpl = document.createElement('template');
    tpl.innerHTML = html;
    let changed = false;
    (function walk(node) {
        Array.from(node.childNodes).forEach(ch => {
            if (ch.nodeType === 1) {
                const cl = ch.classList;
                if (cl.contains('edu-tip') || cl.contains('mytooltip') || cl.contains('mytooltiptext') || cl.contains('xpl')) return;
                walk(ch);
            } else if (ch.nodeType === 3) {
                const frag = _explWrapText(ch.nodeValue);
                if (frag) { ch.replaceWith(frag); changed = true; }
            }
        });
    })(tpl.content);
    return changed ? tpl.innerHTML : html;
};

// Hover display — reuses Code 2's #tooltip-portal element (and this file's
// portal CSS), creating it only if Code 2 hasn't. The two hover systems don't
// fight: each one's mouseover ignores the other's spans, and each mousemove
// only acts while its own span is hovered.
var _eduHover = null;
function _eduPortal() {
    let p = document.getElementById('tooltip-portal');
    if (!p) { p = document.createElement('div'); p.id = 'tooltip-portal'; document.body.appendChild(p); }
    return p;

document.addEventListener('mouseover', (e) => {
    if (!_eduTipsOn) return;
    const t = e.target.closest('.edu-tip');
    if (t === _eduHover) return;
    _eduHover = t;
    if (t) {
        const inner = t.querySelector('.edu-tiptext');
        if (inner) _eduPortal().innerHTML = inner.innerHTML;
    } else {
        _eduPortal().style.display = 'none';
    }
});
document.addEventListener('mousemove', (e) => {
    if (!_eduHover || !_eduTipsOn) return;
    const p = _eduPortal();
    p.style.display = 'block';
    const pad = 16, tw = p.offsetWidth, th = p.offsetHeight;
    let x = e.clientX + pad, y = e.clientY + pad;
    if (x + tw > window.innerWidth)  x = e.clientX - tw - pad;
    if (y + th > window.innerHeight) y = e.clientY - th - pad;
    p.style.left = x + 'px';
    p.style.top  = y + 'px';
});
document.addEventListener('mouseleave', () => {
    if (_eduHover) { _eduPortal().style.display = 'none'; _eduHover = null; }
}, true);
